\chapter{Chosen Attitude Estimation Algorithms}

\section{Mahony filter}

The Mahony filter is a name commonly used to refer to a group of complementary filters derived by Robert Mahony in \cite{mahony-comp-so3}. The main idea behind those algorithms is to provide a computationally inexpensive solution for estimating attitude from inexpensive MEMS IMU sensors. Mahony \textit{et. al.} expands on the idea of the classic complementary filter by introducing a feedback term to the gyroscope measurement, which is an integral of the error between the accelerometer measurement and the expected gravity vector in the sensor frame. This feedback term helps to correct the drift in the gyroscope based estimate, while still maintaining the computational simplicity of a complementary filter.
In this work, the \textit{explicit complementary filter} version of the algorithm proposed in \cite{mahony-comp-so3} will be used.

To derive the correction term for the filter, Madgwick uses an error that is the relative rotation from the body frame to the estimator frame. In practice, the real body frame rotation is unknown, so instead an estimated rotation derived from available measurements $R_{y}$ is used.

\begin{equation}
	E_{tr} = \frac{1}{2} tr(\mathbf{I}_3 - \hat{\mathbf{R}}^T\mathbf{R_y}) = 1 - \cos(\theta)
\end{equation}
Where $\theta$ is the angle associated with the rotation between those frames.
The filter dynamics, in quaternion form, is given by:
\begin{equation}
	\begin{aligned}
		\dot{\mathbf{\hat{q}}}     & = \frac{1}{2} \mathbf{\hat{q}} \otimes \mathbf{p}(\boldsymbol{\Omega}_y - \mathbf{\hat{b}} + k_p \boldsymbol{\omega}) \\
		\boldsymbol{\dot{\hat{b}}} & = -k_I \boldsymbol{\omega}
	\end{aligned}
\end{equation}

The goal of the estimator will be to find a term $\boldsymbol{\omega}$ that will drive this error to 0. The correction term introduced by Mahony \textit{et. al.} is:
\begin{equation}
	\boldsymbol{\omega} = vex(\mathbb{P}_a(\hat{\mathbf{R}}^T\mathbf{R}_y)) = sin(\theta)\mathbf{u}
\end{equation}
With normalized accelerometer measurement $v$ and expected gravity vector $\hat{v} = \mathbf{R}(\mathbf{\hat{q}})^T\mathbf{g}$, $g = [0, 0, 1]^T$, the correction term can be expressed as:
\begin{equation}
	\boldsymbol{\omega} = v \times \hat{v}
\end{equation}

With the correction term defined, the full explicit complementary filter dynamics can be implemented as follows:
\begin{equation}
	\begin{aligned}
		\dot{\hat{q}}   & = \frac{1}{2} \hat{q} \otimes {p}(\Omega_y - \hat{b} + k_p \omega) \\
		{\dot{\hat{b}}} & = -k_I \omega                                                      \\
		{\omega}        & = k_I(v \times \hat{v})
	\end{aligned}
\end{equation}

The estimator can be parameterized by two gains, $k_p$ and $k_I$. The parameter $k_p$ controls the instantaneous   filter correction. If its too high, the filter will be very sensitive to accelerometer noise, while if its too low, the filter will be slow to correct for gyroscope drift, or even diverge all together. The parameter $k_I$ controls the integral correction term, which is responsible for estimating the gyroscope bias.

\section{AQUA Complementary filter}

In \cite{Valenti-AQUA} Roberto G. Valenti \textit{et. al.} describe an orientation estimation algorithm, which is an interesting variation on a standard complementary filter algorithm. The main idea behind his approach is to provide an estimation as the algebraic solution of a system from inertial and optionally magnetic observations. % Napisać bardziej po swojemu?
During the correction step, the algorithm finds the tilt quaternion from the accelerometer and the heading quaternion from the magnetometer observations separately. This allows to avoid the problem of corrections originating from the magnetometer measurements affecting the attitude estimation.

%
The dynamics of the filter prediction step, as proposed in \cite{Valenti-AQUA}, but adjusted for notation conventions, is given by:
\begin{equation}
	\begin{aligned}
		{}^{B}_{E}\dot{\hat{\boldsymbol{q}}}_{\omega,t_k} & = \frac{1}{2} \hat{\boldsymbol{q}}_{\omega,t_{k - 1}} \otimes \boldsymbol{p}(\boldsymbol{\Omega}_{y,t_k}) \\
	\end{aligned}
\end{equation}
This equation can be computed numerically, using a simple Euler integration method:
\begin{equation}
	{}^{B}_{E}\hat{\boldsymbol{q}}_{\omega,t_k} = \hat{\boldsymbol{q}}_{\omega,t_{k - 1}} + \dot{\hat{\boldsymbol{q}}}_{\omega,t_k} \Delta t
\end{equation}
In the correction step, a delta quaternion is computed from the accelerometer observation, which is then used to correct the predicted orientation.
\begin{equation}
	\hat{\boldsymbol{q}}_{t_k} = {}^{E}_{W}\widehat{\Delta\boldsymbol{q}}_{acc} \otimes {}^{B}_{E}\hat{\boldsymbol{q}}_{\omega,t_k}
\end{equation}
The delta quaternion $\Delta \boldsymbol{q}_{acc}$satisfies the following system of equations:
% Ogarnij to równanie, bo tu znaki się nie będą zgadzać
\begin{equation}
	\begin{cases}
		2(\Delta q_{1, acc}\Delta q_{3, acc} + \Delta q_{0, acc}\Delta q_{2, acc})                & = G_x \\
		2(\Delta q_{2, acc}\Delta q_{3, acc} + \Delta q_{0, acc}\Delta q_{1, acc})                & = G_y \\
		\Delta q_{0, acc}^2 - \Delta q_{1, acc}^2 - \Delta q_{2, acc}^2    +  \Delta q_{3, acc}^2 & = G_z \\
	\end{cases}
\end{equation}
Where $\boldsymbol{G} = \boldsymbol{R}({}^{B}_{W}\hat{\boldsymbol{q}})\boldsymbol{g} = [G_x, G_y, G_z]^T, g = [0, 0, 1]^T$, is the gravity vector predicted from the orientation estimate.
After defining the component $\Delta q_{3, acc} = 0$ or $\Delta q_{2, acc} = 0$, the solution to this system of can be found, one for each case. Two solutions were given to avoid singularities in the solution, depending on the value of $G_z$:
\begin{equation}
	\Delta \boldsymbol{q}_{acc} = \begin{cases}
		[ \sqrt{\frac{G_z + 1}{2}} \quad \frac{G_y}{2G_z + 1} \quad -\frac{G_x}{2G_z + 1} \quad 0 ]  & , G_z >= 0 \\
		[ -\frac{G_y}{2(1-G_z)} \quad -\sqrt{\frac{1-G_z}{2}	} \quad 0 \quad -\frac{G_x}{2(1-G_z)} ] & , G_z < 0  \\
	\end{cases}
\end{equation}
%
Applying the full correction term will instantaneously rotate the predicted orientation to match the tilt observed by the accelerometer. This is undesirable, as the accelerometer measures both the gravity vector and the linear acceleration of the sensor. To avoid this, $\Delta \boldsymbol{q}_{acc}$ is interpolated with the identity quaternion, using a gain parameter $k_{acc}$. Valenti \textit{et. al.} propose using LERP interpolation for relatively small corrections. The metric given for this selection is the cosine of the subtended angle $\Omega$ between the interpolated quaternions. The suggested threshold over which  \textit{LERP} should be used is $cos(\Omega) > 0.9$, \textit{SLERP} should be used otherwise. When interpolating with a quaternion, $cos(\Omega)$ is equal to the scalar part of the $\Delta \boldsymbol{q}_{acc}$ quaternion. The final correction term becomes:
\begin{equation}
	{}^{E}_{W}\widehat{\Delta\boldsymbol{q}}_{acc} = \begin{cases}
		\frac{\textit{LERP}(\boldsymbol{q}_I, \Delta \boldsymbol{q}_{acc}, k_{acc})}{||\textit{LERP}(\boldsymbol{q}_I, \Delta \boldsymbol{q}_{acc}, k_{acc})||} & , \Delta q_{0, acc} > 0.9    \\
		\textit{SLERP}(\boldsymbol{q}_I,\Delta \boldsymbol{q}_{acc}, k_{acc})                                                                                   & , \Delta q_{0, acc} \leq 0.9 \\
	\end{cases}
\end{equation}

\section{Madgwick filter}

In \cite{Madgwick2010AnEO} Sebastian O.H. Madgwick presents an estimation algorithm that departs from the traditional observer design, and threats the problem of attitude estimation, directly as an optimization problem. The algorithm uses the accelerometer and magnetometer measurements in an analytically derived gradient descent algorithm, minimizing the gyroscope measurement errors.

The optimization problem from IMU observations is defined as follows:
\begin{equation}
	\hat{\boldsymbol{q}}_{k+1} = \hat{\boldsymbol{q}}_{k} - \mu\frac{\nabla f(\hat{\boldsymbol{q}}_{k}, \boldsymbol{a}_y)}{||\nabla f(\hat{\boldsymbol{q}}_{k}, \boldsymbol{a}_y)||}
\end{equation}

\begin{equation}
	\nabla f(\hat{\boldsymbol{q}}_{k}, \boldsymbol{a}_y) = \boldsymbol{J}^T(\boldsymbol{q}_{k}) f(\hat{\boldsymbol{q}}_{k}, \boldsymbol{a}_y)
\end{equation}

\begin{equation}
	f(\boldsymbol{q}, {}^B\boldsymbol{a}) = \begin{bmatrix}
		2(q_1q_3 - q_0q_2) - a_{x} \\
		2(q_0q_1 + q_2q_3) - a_{y} \\
		2(\frac{1}{2} - q_1^2 - q_2^2) - a_{z}\
	\end{bmatrix}
\end{equation}

\begin{equation}
	\boldsymbol{J}(\boldsymbol{q}) = \begin{bmatrix}
		-2q_2 & 2q_3  & -2q_0 & 2q_1 \\
		2q_1  & 2q_0  & 2q_3  & 2q_2 \\
		0     & -4q_1 & -4q_2 & 0    \\
	\end{bmatrix}
\end{equation}
Using this optimization problem, the filter dynamics can be defined as:
\begin{equation}
	\begin{aligned}
		\hat{\boldsymbol{q}}_{est,t}             & = \hat{\boldsymbol{q}}_{est,t-1} + \dot{\boldsymbol{q}}_{est,t}\Delta t                                                                \\
		\dot{\boldsymbol{q}}_{est,t}             & = \dot{\boldsymbol{q}}_{\omega,t} - \beta \dot{\hat{\boldsymbol{q}}}_{\epsilon, t}                                                     \\
		\dot{\hat{\boldsymbol{q}}}_{\epsilon, t} & = \frac{\nabla f(\hat{\boldsymbol{q}}_{est,t- 1}, \boldsymbol{a}_y)}{||\nabla f(\hat{\boldsymbol{q}}_{est,t - 1}, \boldsymbol{a}_y)||} \\
	\end{aligned}
\end{equation}

\section{Extended Kalman Filter}

The Extended Kalman Filter is a generalization of the classic linear Kalman observer, that allows it to be applied to nonlinear systems. It is achieved by linearizing the system dynamics and measurement equations around the current state estimate, and then applying the standard Kalman filter equations to the linearized system.

The discrete observer state transition and observation models are defined as follows:
\begin{equation}
	\begin{aligned}
		\boldsymbol{x}_{k} & = f(\boldsymbol{x}_{k-1}, \boldsymbol{u}_{k-1}) + \boldsymbol{w}_{k-1} \\
		\boldsymbol{z}_{k} & = h(\boldsymbol{x}_{k}) + \boldsymbol{v}_{k}                           \\
	\end{aligned}
\end{equation}

EKF is a recursive algorithm, which consists of two steps: prediction and correction (or innovation). In the prediction step, the state estimate and error covariance are propagated forward in time using the system dynamics. In the correction step, the state estimate and its error covariance are corrected using the new measurement and calculated Kalman gain.

Prediction:
\begin{equation}
	\begin{aligned}
		\hat{\boldsymbol{x}}_{k|k-1} & = f(\hat{\boldsymbol{x}}_{k-1|k-1}, \boldsymbol{u}_{k-1})                                            \\
		\boldsymbol{P}_{k|k-1}       & = \boldsymbol{F}_k \boldsymbol{P}_{k-1|k-1} \boldsymbol{F}_k^T + \boldsymbol{Q}_{k -1}               \\
		\boldsymbol{F}_k             & = \frac{\partial f}{\partial \boldsymbol{x}}|_{\hat{\boldsymbol{x}}_{k-1|k-1}, \boldsymbol{u}_{k-1}}
	\end{aligned}
\end{equation}

Where:
\begin{eqwhere}[3cm]
	\item[$\boldsymbol{P}_{k|k-1}  $] Predicted covariance of the state estimate
	\item[$\boldsymbol{F}_k$] State transition Jacobian
	\item[$\boldsymbol{Q}_{k-1}$] Process noise covariance matrix
\end{eqwhere}

Innovation:
\begin{equation}
	\begin{aligned}
		\tilde{\boldsymbol{y}}_k   & = \boldsymbol{z}_k - h(\hat{\boldsymbol{x}}_{k|k-1})                            \\
		\boldsymbol{S}_k           & = \boldsymbol{H}_k \boldsymbol{P}_{k|k-1} \boldsymbol{H}_k^T + \boldsymbol{R}_k \\
		\boldsymbol{K}_k           & = \boldsymbol{P}_{k|k-1} \boldsymbol{H}_k^T \boldsymbol{S}_k^{-1}               \\
		\hat{\boldsymbol{x}}_{k|k} & = \hat{\boldsymbol{x}}_{k|k-1} + \boldsymbol{K}_k \tilde{\boldsymbol{y}}_k      \\
		\boldsymbol{P}_{k|k}       & = (\mathbf{I} - \boldsymbol{K}_k\boldsymbol{H}_k)\boldsymbol{P}_{k|k-1}         \\
		\boldsymbol{H}_k           & = \frac{\partial h}{\partial \boldsymbol{x}}|_{\hat{\boldsymbol{x}}_{k|k-1}}
	\end{aligned}
\end{equation}

Where:
\begin{eqwhere}[3cm]
	\item[$\boldsymbol{y}_k$] Measurement innovation
	\item[$\boldsymbol{S}_k$] Innovation covariance
	\item[$\boldsymbol{H}_k$] Observation model Jacobian
	\item[$\boldsymbol{R}_k$] Measurement noise covariance matrix
	\item[$\boldsymbol{K}_k$] Near-optimal Kalman gain
\end{eqwhere}

\newpage

When applying the EKF to the problem of attitude estimation, the gyroscope measurements are used as the model input, while the accelerometer and optionally magnetometer measurements are used for innovation.
The state vector $\boldsymbol{x}$ and the input vector $\boldsymbol{u}$ can be defined as follows:
%
\begin{equation}
	\boldsymbol{x} = \boldsymbol{q} = \left[ q_0\ q_1\ q_2\ q_3 \right]^T
\end{equation}
%
\begin{equation}
	\boldsymbol{u} = \boldsymbol{\Omega}_{y} = \left[ \omega_{x}, \omega_{y}, \omega_{z} \right]^T
\end{equation}
%
The state transition function and its Jacobian can be defined using \eqref{eq:gyro_q_dot}
\begin{equation}
	f(\boldsymbol{x}_{k}, \boldsymbol{u}_{k}) = \boldsymbol{q}_{k} + \frac{1}{2}\boldsymbol{q}_{k} \otimes \boldsymbol{p}(\boldsymbol{\Omega}_{y,k})\Delta t
\end{equation}

\begin{equation}
	\boldsymbol{F}(\boldsymbol{x}_{k}, \boldsymbol{u}_{k}) = \begin{bmatrix}
		1                               & -\frac{1}{2}\omega_{x,k}\Delta t & -\frac{1}{2}\omega_{y,k}\Delta t & -\frac{1}{2}\omega_{z,k}\Delta t \\
		\frac{1}{2}\omega_{x,k}\Delta t & 1                                & \frac{1}{2}\omega_{z,k}\Delta t  & -\frac{1}{2}\omega_{y,k}\Delta t \\
		\frac{1}{2}\omega_{y,k}\Delta t & -\frac{1}{2}\omega_{z,k}\Delta t & 1                                & \frac{1}{2}\omega_{x,k}\Delta t  \\
		\frac{1}{2}\omega_{z,k}\Delta t & \frac{1}{2}\omega_{y,k}\Delta t  & -\frac{1}{2}\omega_{x,k}\Delta t & 1
	\end{bmatrix}
\end{equation}

With IMU only attitude estimation, the observation $\boldsymbol{z}_k$ represent the accelerometer measurements in the body frame.
The measurement estimate can be computed by rotating the global gravity vector ${}^{W}\boldsymbol{g} = [0\ 0\ 1]^T$ into the body frame using the current orientation estimate $\hat{\boldsymbol{q}}_{k|k-1}$:
\begin{equation}
	\begin{aligned}
		\tilde{y}_k         & = \boldsymbol{a}_y - h(\boldsymbol{q}_k)                       \\
		h(\boldsymbol{q}_k) & = \mathbf{R}^T(\hat{\boldsymbol{q}}_{k})\ {}^{W}\boldsymbol{g} \\
	\end{aligned}
\end{equation}
The Jacobian of the observation model is:
\begin{equation}
	\boldsymbol{H}(\boldsymbol{x}_k) = \begin{bmatrix}
		-2q_2 & 2q_3  & -2q_0 & 2q_1 \\
		2q_1  & 2q_0  & 2q_3  & 2q_2 \\
		2q_0  & -2q_1 & -2q_2 & 2q_3
	\end{bmatrix}
\end{equation}
